  <h1>🧠 Constructeur de Prompt ACTIF</h1>

  <label for="identite">🔷 Identité (Qui doit répondre ?)</label>
  <input id="identite" type="text" placeholder="Ex : Agis comme un expert en cybersécurité">

  <label for="contexte">🔶 Contexte (À propos de quoi ?)</label>
  <input id="contexte" type="text" placeholder="Ex : Pour une startup qui développe une application mobile">

  <label for="action">🔹 Action (Que doit faire l'IA ?)</label>
  <input id="action" type="text" placeholder="Ex : Donne 5 conseils pour améliorer la sécurité des données">

  <label for="tonalite">🔸 Tonalité (Quel style ?)</label>
  <input id="tonalite" type="text" placeholder="Ex : Avec un ton professionnel et accessible">

  <label for="format">🔺 Format (Sous quelle forme ?)</label>
  <input id="format" type="text" placeholder="Ex : Présente la réponse sous forme de liste à puces">

  <button onclick="genererPrompt()">✅ Générer le prompt</button>

  <div id="result"></div>

  <script>
    function genererPrompt() {
      const i = document.getElementById("identite").value;
      const c = document.getElementById("contexte").value;
      const a = document.getElementById("action").value;
      const t = document.getElementById("tonalite").value;
      const f = document.getElementById("format").value;

      const prompt = `${i}. ${c}. ${a}. ${t}. ${f}.`;
      document.getElementById("result").innerHTML = `
        <h3>📋 Prompt généré :</h3>
        <textarea rows="6" readonly>${prompt}</textarea>
        <button onclick="copierPrompt()">📋 Copier le prompt</button>
      `;
    }

    function copierPrompt() {
      const textarea = document.querySelector("#result textarea");
      textarea.select();
      document.execCommand("copy");
      alert("Prompt copié dans le presse-papiers !");
    }
  </script>
